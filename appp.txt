def ask_qwen(text):
    prompt = f"""
Ты — умный помощник студента.

На основе расшифровки лекции составь понятный учебный конспект.

Используй только информацию из расшифровки.
Не придумывай факты.

Структура:

📌 ТЕМА ЛЕКЦИИ

📝 КРАТКОЕ СОДЕРЖАНИЕ

🔑 КЛЮЧЕВЫЕ ПОНЯТИЯ
- термин — простое объяснение

📚 ОСНОВНЫЕ ТЕЗИСЫ
- главная мысль
- главная мысль
- главная мысль

💡 ВАЖНЫЕ ФАКТЫ И ПРИМЕРЫ

🎯 ЧТО НУЖНО ЗАПОМНИТЬ

⚡ ПОВТОРЕНИЕ ЗА 1 МИНУТУ

Пиши на русском языке.

РАСШИФРОВКА:
{text}
"""

    data = json.dumps({
        "model": "qwen2.5:3b",
        "prompt": prompt,
        "stream": False
    }).encode("utf-8")

    request = urllib.request.Request(
        "http://localhost:11434/api/generate",
        data=data,
        headers={"Content-Type": "application/json"}
    )

    with urllib.request.urlopen(request, timeout=600) as response:
        result = json.loads(
            response.read().decode("utf-8")
        )

    return result["response"]

def create_pdf(text, filename="lecture_notes.pdf"):

    pdfmetrics.registerFont(
        TTFont(
            "Arial",
            "C:/Windows/Fonts/arial.ttf"
        )
    )
    styles = getSampleStyleSheet()

    styles["Normal"].fontName = "Arial"
    styles["Normal"].fontSize = 10
    styles["Normal"].leading = 14

    styles["Title"].fontName = "Arial"

    document = SimpleDocTemplate(
        filename,
        pagesize=A4,
        rightMargin=40,
        leftMargin=40,
        topMargin=40,
        bottomMargin=40
    )

    elements = []

    elements.append(
        Paragraph(
            "🎓 LectureAI — конспект лекции",
            styles["Title"]
        )
    )

    elements.append(
        Spacer(1, 20)
    )

    for line in text.split("\n"):

        line = line.strip()

        if line:

            line = line.replace(
                "&",
                "&amp;"
            )

            elements.append(
                Paragraph(
                    line,
                    styles["Normal"]
                )
            )

            elements.append(
                Spacer(1, 8)
            )

    document.build(elements)

    return filename

st.markdown("## 🎙️ Загрузка лекции")

st.caption(
    "Поддерживаются форматы MP3, WAV, M4A и OGG"
)

uploaded_file = st.file_uploader(
    "Выберите аудиофайл",
    type=["mp3", "wav", "m4a", "ogg"],
    label_visibility="collapsed"
)

if uploaded_file is not None:

    st.success(f"Файл загружен: {uploaded_file.name}")

    st.audio(
        uploaded_file,
        format=uploaded_file.type
    )

    if st.button(
    "🚀 Расшифровать лекцию",
    use_container_width=True
):

        temp_path = None

        try:

            with st.spinner("🎧 Распознаю речь..."):

                suffix = Path(
                    uploaded_file.name
                ).suffix

                with tempfile.NamedTemporaryFile(
                    delete=False,
                    suffix=suffix
                ) as temp_file:

                    temp_file.write(
                        uploaded_file.getbuffer()
                    )

                    temp_path = temp_file.name

                model = load_whisper()

                segments, info = model.transcribe(
                    temp_path,
                    language="ru",
                    beam_size=5
                )

                text = " ".join(
                    segment.text.strip()
                    for segment in segments
                )

            st.session_state["lecture_text"] = text

            st.success("✅ Лекция расшифрована!")

        except Exception as error:

            st.error(f"❌ Ошибка распознавания: {error}")

        finally:

            if temp_path is not None:
                if os.path.exists(temp_path):
                    os.remove(temp_path)

if "test" in st.session_state:

    st.subheader("📝 Тест по лекции")

    test_text = st.session_state["test"]

    questions = []
    blocks = test_text.split("ВОПРОС ")

    for block in blocks[1:]:

        lines = block.strip().split("\n")

        question_text = ""
        options = {}
        correct = ""

        for line in lines:

            line = line.strip()

            if line.startswith(("1:", "2:", "3:", "4:", "5:")):
                question_text = line

            elif line.startswith("A)"):
                options["A"] = line[2:].strip()

            elif line.startswith("B)"):
                options["B"] = line[2:].strip()

            elif line.startswith("C)"):
                options["C"] = line[2:].strip()

            elif line.startswith("D)"):
                options["D"] = line[2:].strip()

            elif line.startswith("ПРАВИЛЬНЫЙ:"):
                correct = line.split(":")[-1].strip()

        if question_text and len(options) == 4 and correct:
            questions.append({
                "question": question_text,
                "options": options,
                "correct": correct
            })

    answers = {}

    for i, q in enumerate(questions):

        st.markdown(f"### {i + 1}. {q['question']}")

        answers[i] = st.radio(
            "Выберите ответ:",
            list(q["options"].keys()),
            format_func=lambda x, opts=q["options"]: f"{x}) {opts[x]}",
            key=f"question_{i}"
        )

    if st.button("✅ Проверить тест"):

        score = 0

        for i, q in enumerate(questions):

            if answers[i] == q["correct"]:
                score += 1

        st.success(
            f"🎉 Результат: {score} из {len(questions)}"
        )
